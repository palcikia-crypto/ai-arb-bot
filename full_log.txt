def ai_analyze(title, description):
    text = (title + " " + description).lower()
    region = "UNKNOWN"
    if any(word in text for word in ["турция", " tr ", " try ", "турецк"]):
        region = "Turkey"
    elif any(word in text for word in ["сша", " us ", " usd ", "global", "весь мир"]):
        region = "Global"
    elif "аргентин" in text:
        region = "Argentina"

    is_risky = False
    risk_reason = ""
    if region == "Turkey" and "весь мир" in text:
        is_risky = True
        risk_reason = "Обман: Турецкая карта не может быть 'весь мир'"
    
    return {"region": region, "is_risky": is_risky, "risk_reason": risk_reason}

def calculate_arbitrage(buy_price_usd, face_value_usd, title, description):
    analysis = ai_analyze(title, description)
    region = analysis["region"]
    discount_ratio = buy_price_usd / face_value_usd
    carding_risk = discount_ratio < 0.65
    results = []
    
    for buyer_name, rates in BUYERS.items():
        rate = rates.get(region, 0)
        if rate == 0: continue
            
        sell_price = face_value_usd * rate
        profit = sell_price - buy_price_usd
        margin = (profit / buy_price_usd) * 100
        
        if analysis["is_risky"]:
            status = "⚠️ СКАМ (Обман региона)"
        elif carding_risk:
            status = "❌ РИСК (Кардинг)"
        elif profit > 0:
            status = "✅ ПРОФИТ"
        else:
            status = "🔻 УБЫТОК"
            
        results.append({
            "buyer": buyer_name, "region": region, "rate": rate,
            "profit": round(profit, 2), "margin": round(margin, 1),
            "status": status, "ai_reason": analysis["risk_reason"]
        })
    return results

HEADERS = {"User-Agent": "Mozilla/5.0 (Windows NT 10.0; Win64; x64) AppleWebKit/537.36 Chrome/120.0.0.0 Safari/537.36"}

def web_search(query, max_results=5):
    try:
        r = requests.post("https://html.duckduckgo.com/html/", data={"q": query}, headers=HEADERS, timeout=15)
        if r.status_code != 200: return f"SEARCH_ERROR: HTTP {r.status_code}"
        soup = BeautifulSoup(r.text, "html.parser")
        results = []
        for res in soup.select(".result")[:max_results]:
            a = res.select_one(".result__a")
            snip = res.select_one(".result__snippet")
            if a:
                href = a.get("href", "")
                m = re.search(r'uddg=([^&]+)', href)
                if m: href = unquote(m.group(1))
                title = a.get_text(strip=True)
                body = snip.get_text(strip=True) if snip else ""
                results.append(f"URL: {href}\n  {title}: {body[:150]}")
        return "\n".join(results) if results else "НЕТ РЕЗУЛЬТАТОВ"
    except Exception as e:
        return f"SEARCH_ERROR: {e}"

def web_fetch(url, max_chars=4000):
    try:
        r = requests.get(url, headers=HEADERS, timeout=15)
        if r.status_code != 200: return f"FETCH_ERROR: HTTP {r.status_code}"
        soup = BeautifulSoup(r.text, "html.parser")
        for s in soup(["script", "style", "noscript"]): s.decompose()
        text = soup.get_text(separator=" ", strip=True)
        text = re.sub(r'\s+', ' ', text)
        return text[:max_chars]
    except Exception as e:
        return f"FETCH_ERROR: {e}"

def llm_extract_unit_price(text, item_name):
    """LLM извлекает цену ЗА 1 ЕДИНИЦУ (делит цену пакета на количество)."""
    if not HAS_LLM: return None
    prompt = f"""Текст страницы:
{text[:4000]}

ЗАДАЧА: найди цену за 1 единицу товара '{item_name}' в USD.
Если на странице есть пакеты (например "200 Robux = $2.25"), вычисли цену за 1 Robux: 2.25 / 200 = 0.01125.
Если цены в рублях — переведи в USD по курсу 90.
Если цены нет — ответь NONE.

Ответ: ТОЛЬКО число (например 0.01125) или NONE."""
    try:
        ans = ask(prompt, system_prompt="Отвечай только числом или NONE.")
        ans = ans.strip().replace(",", ".")
        m = re.search(r'\d+\.\d+', ans)
        return float(m.group(0)) if m else None
    except: return None

def find_price_via_search(item_name, search_query):
    """Поиск + fetch + LLM extraction. Возвращает (price_usd, url)."""
    print(f"  🔍 Поиск: {search_query}")
    search_result = web_search(search_query)
    urls = re.findall(r'URL: (\S+)', search_result)
    print(f"  → Найдено URL: {len(urls)}")
    
    for i, url in enumerate(urls[:3]):
        print(f"  📥 Читаем [{i+1}/3]: {url[:80]}")
        page_text = web_fetch(url)
        if page_text.startswith("FETCH_ERROR"): continue
        
        price = llm_extract_unit_price(page_text, item_name)
        if price and 0.0001 < price < 100:
            print(f"    ✅ Цена: ${price:.5f}")
            return price, url
        else:
            print(f"    ❌ Цена не найдена")
    return None, None

def find_context(text, keywords, window=300):
    """Ищет ключевые слова и возвращает контекст."""
    results = []
    text_lower = text.lower()
    for kw in keywords:
        idx = 0
        while True:
            idx = text_lower.find(kw, idx)
            if idx == -1: break
            context = text[max(0, idx-window):idx+window]
            # Чистим от HTML-тегов
            context = re.sub(r'<[^>]+>', ' ', context)
            context = re.sub(r'\s+', ' ', context).strip()
            if len(context) > 50:
                results.append(f"[{kw}] {context}")
            idx += len(kw)
            if len(results) > 10: break
        if len(results) > 10: break
    return results

def analyze_file(filename, platform):
    print(f"\n{'='*60}")
    print(f"  📦 {platform} — {filename}")
    print(f"{'='*60}")
    
    try:
        data = json.load(open(filename))
    except:
        print(f"  ❌ Файл не найден")
        return
    
    # Собираем весь текст из всех полей
    all_text = ""
    for key, val in data.items():
        if key == "platform": continue
        if isinstance(val, str):
            all_text += f"\n\n=== {key} ===\n{val}"
        elif isinstance(val, dict):
            for k, v in val.items():
                all_text += f"\n\n=== {key}.{k} ===\n{v}"
        elif isinstance(val, list):
            for item in val:
                if isinstance(item, dict):
                    all_text += f"\n{item.get('title','')} — {item.get('snippet','')}"
    
    print(f"  📊 Всего текста: {len(all_text)} символов")
    
    # Ищем ключевые слова
    keywords = ["turkey", "turkish", "steam turkey", "85%", "86%", "rate", "steam"]
    contexts = find_context(all_text, keywords)
    
    if contexts:
        print(f"\n  🔍 Найдено {len(contexts)} контекстов:")
        for c in contexts[:8]:
            print(f"  {c[:200]}")
    else:
        print(f"  ⚠️ Ничего не найдено")
    
    # ИИ анализирует
    if contexts:
        prompt = f"""Площадка: {platform}
Вот фрагменты текста с их сайта (найденные по ключевым словам):

{chr(10).join(contexts[:10])[:3000]}

Ответ СТРОГО:
ПРИНИМАЕТ_TURKEY: ДА / НЕТ / НЕИЗВЕСТНО

ЦИТАТА: <цитата из текста>"""
        
        try:

            print(f"  {ans[:500]}")

def parse_prices(html, patterns):
    if not html: return []

    return results

def ai_analyze(title, description):
    text = (title + " " + description).lower()
    region = "UNKNOWN"
    if any(word in text for word in ["турция", " tr ", " try ", "турецк"]):
        region = "Turkey"
    elif any(word in text for word in ["сша", " us ", " usd ", "global", "весь мир"]):
        region = "Global"
    elif "аргентин" in text:
        region = "Argentina"

    is_risky = False
    risk_reason = ""
    if region == "Turkey" and "весь мир" in text:
        is_risky = True
        risk_reason = "Обман: Турецкая карта не может быть 'весь мир'"
    
    return {"region": region, "is_risky": is_risky, "risk_reason": risk_reason}

def calculate_arbitrage(buy_price_usd, face_value_usd, title, description):
    analysis = ai_analyze(title, description)
    region = analysis["region"]
    discount_ratio = buy_price_usd / face_value_usd
    carding_risk = discount_ratio < 0.65
    results = []
    
    for buyer_name, rates in BUYERS.items():
        rate = rates.get(region, 0)
        if rate == 0: continue
            
        sell_price = face_value_usd * rate
        profit = sell_price - buy_price_usd
        margin = (profit / buy_price_usd) * 100
        
        if analysis["is_risky"]:
            status = "⚠️ СКАМ (Обман региона)"
        elif carding_risk:
            status = "❌ РИСК (Кардинг)"
        elif profit > 0:
            status = "✅ ПРОФИТ"
        else:
            status = "🔻 УБЫТОК"
            
        results.append({
            "buyer": buyer_name, "region": region, "rate": rate,
            "profit": round(profit, 2), "margin": round(margin, 1),
            "status": status, "ai_reason": analysis["risk_reason"]
        })
    return results

def try_jina_reader(url):

def try_mobile_ua(url):
    """ЛАЙФХАК 2: мобильный User-Agent — часто отдаёт простой HTML."""

    try:
        r = requests.get(url, headers=MOBILE_HEADERS, timeout=15)
        if r.status_code == 200:
            return r.text
    except: pass
    return None

def try_wayback(url):
    """ЛАЙФХАК 3: Wayback Machine — архивные копии (обход 403)."""
    print(f"  [Wayback] {url}")
    try:
        r = requests.get(f"http://archive.org/wayback/available?url={url}", timeout=15)
        data = r.json()
        snapshot = data.get("archived_snapshots", {}).get("closest", {})
        if snapshot.get("url"):
            print(f"    → snapshot: {snapshot['url']}")
            r2 = requests.get(snapshot["url"], headers=HEADERS, timeout=15)
            if r2.status_code == 200:
                return r2.text
    except Exception as e:
        print(f"    ⚠️ {e}")
    return None

def find_next_data(html):
    """ЛАЙФХАК 4: __NEXT_DATA__ (Next.js) — данные встроены в HTML."""
    if not html: return None
    m = re.search(r'<script id="__NEXT_DATA__" type="application/json">(.*?)</script>', html, re.DOTALL)
    if m:
        try:
            return json.loads(m.group(1))
        except: pass
    return None

def find_api_endpoints(html):
    """ЛАЙФХАК 5: поиск API endpoints в JS-бандлах."""
    if not html: return []
    endpoints = set()
    # 1. Из HTML
    for m in re.finditer(r'["\'](/(?:api|v\d+)/[\w/\-\.]+)["\']', html):
        endpoints.add(m.group(1))
    # 2. Из JS-файлов
    js_urls = re.findall(r'<script[^>]+src="([^"]+\.js[^"]*)"', html)
    for js in js_urls[:3]:
        if not js.startswith("http"):
            js = "https://" + js.lstrip("/")
        try:
            r = requests.get(js, headers=HEADERS, timeout=10)
            for m in re.finditer(r'["\'](/(?:api|v\d+)/[\w/\-\.]+)["\']', r.text):
                endpoints.add(m.group(1))
        except: pass
    return list(endpoints)[:10]

def try_api_endpoints(domain, endpoints):
    """ЛАЙФХАК 6: пробуем найденные API endpoints на Turkey."""
    results = {}
    for ep in endpoints:
        for proto in ["https://"]:
            for dom in [domain]:
                url = f"{proto}{dom}{ep}"
                try:
                    r = requests.get(url, headers=HEADERS, timeout=10)
                    if r.status_code == 200 and len(r.text) > 100:
                        # Ищем Turkey
                        text_lower = r.text.lower()
                        if "turkey" in text_lower or "turkish" in text_lower or "steam" in text_lower:
                            results[url] = r.text[:1500]
                except: continue
    return results

def fetch_plati(query):
    params = {"query": query, "pagesize": 50, "pagenum": 1,
              "visibleOnly": "true", "response": "json"}
    try:
        r = requests.get(API, params=params, headers=HEADERS, timeout=20)
        return r.json().get("items", [])
    except: return []

def find_sane_lot(items, target, price_min, price_max):
    """Ищет лот, где target в названии И цена в разумных пределах."""
    candidates = []
    for item in items:
        name = (item.get("name_eng","") + " " + item.get("name","")).lower()
        price = item.get("price_usd")
        url = item.get("url")
        if not price or not url: continue
        
        # Убираем мусор
        name_clean = re.sub(r'\[\s*-?\d+%?\s*\]', '', name)
        name_clean = re.sub(r'-\d+%', '', name_clean)
        
        # Ищем target в названии (как отдельное число)
        if not re.search(rf'\b{target}\b', name_clean):
            continue
        
        # ЦЕНА в разумных пределах
        if not (price_min <= price <= price_max):
            print(f"  ⚠️ Отброшен: {name[:50]} | ${price:.2f} (цена не в {price_min}-{price_max})")
            continue
        
        candidates.append({
            "price": float(price),
            "seller": item.get("seller_name", ""),
            "url": url,
            "name": item.get("name", "")[:60],
        })
    candidates.sort(key=lambda x: x["price"])
    return candidates

def fetch(url, timeout=15):
    try:
        r = requests.get(url, headers=HEADERS, timeout=timeout)
        if r.status_code != 200:
            return None, r.status_code
        soup = BeautifulSoup(r.text, "html.parser")
        for s in soup(["script", "style", "noscript"]):
            s.decompose()
        text = soup.get_text(separator=" ", strip=True)
        text = re.sub(r'\s+', ' ', text)
        return text[:4000], 200
    except Exception as e:

    return None

def get_lots_for_nominal(nominal):
    """Возвращает ВСЕ доступные лоты для номинала, где nominal = минимальный."""
    params = {
        "query": f"Steam Turkey {nominal} USD",
        "pagesize": 50,
        "pagenum": 1,
        "visibleOnly": "true",
        "response": "json"
    }
    try:
        r = requests.get(API, params=params, headers=HEADERS, timeout=20)
        items = r.json().get("items", [])
    except Exception as e:
        print(f"  ⚠️ API error {nominal}: {e}")
        return []
    
    lots = []
    for item in items:
        name = item.get("name_eng", "").lower() + " " + item.get("name", "").lower()
        price = item.get("price_usd")
        url = item.get("url")
        if not price or not url:
            continue
        
        # Первое число в названии
        m = re.search(r'(\d+)', name)
        if not m:
            continue
        first_num = int(m.group(1))
        
        # Только если первый номинал = искомый
        if first_num != nominal:
            continue
        
        # Отсеиваем подозрительно дешёвые (обычно цена за минимальный в диапазоне)
        if price < nominal * 0.4:
            continue
        
        lots.append({
            "nominal": nominal,
            "price": float(price),
            "seller": item.get("seller_name", ""),
            "url": url,
            "name": item.get("name", "")[:60],
        })
    
    lots.sort(key=lambda x: x["price"])
    return lots

def scan():

def ai_analyze(title):
    """ИИ проверяет название на обман и кардинг"""
    text = title.lower()
    is_scam = False
    reason = ""
    
    # Если это Турция, но пишут "весь мир" или "Global" - это скам
    if any(w in text for w in ["турция", " tr ", " try "]):

            is_scam = True

    return is_scam, reason

import sys, re, requests
from pathlib import Path
from bs4 import BeautifulSoup

# Подключаем наш llm_backend
sys.path.append(str(Path.home() / "quantum_lab_sandbox" / "tools" / "agent_dev"))
try:
    from llm_backend import ask
    HAS_LLM = True
except Exception as e:
    HAS_LLM = False
    print(f"⚠️ LLM не подключен: {e}")

HEADERS = {"User-Agent": "Mozilla/5.0 Chrome/120.0"}

# Что мы знаем о скупщиках (региональные ограничения)
BUYER_REGIONS = {
    "GiftCardIo":  "Принимает global/Turkey Steam gift cards (нужно проверить конкретную ставку)",
    "CardBix":     "Принимает Steam gift cards разных регионов (нужно проверить Turkey)",
    "Cashard":     "Принимает Steam gift cards (нужно проверить Turkey)",
    "SellCardNow": "Принимает Steam gift cards (нужно проверить Turkey)",
    "HighCardRate":"Принимает Steam gift cards (нужно проверить Turkey)",
}

def fetch_description(url):
    """Загружает страницу и извлекает описание товара."""
    try:
        r = requests.get(url, headers=HEADERS, timeout=20)
        if r.status_code != 200:
            return None
        soup = BeautifulSoup(r.text, "html.parser")
        # Убираем скрипты/стили
        for s in soup(["script", "style", "noscript"]):
            s.decompose()
        text = soup.get_text(separator=" ", strip=True)
        text = re.sub(r'\s+', ' ', text)
        return text[:3500]
    except Exception as e:
        return None

def rule_based_check(description):
    """Быстрая проверка на явные red flags."""
    d = description.lower()
    flags = []
    if "только для региона турция" in d or "только для турции" in d:
        flags.append("⚠️ Только для региона Турция")
    if "только для аккаунтов с валютой" in d and "usd" in d:
        flags.append("⚠️ Только для USD-аккаунтов")
    if "нельзя использовать" in d and "других стран" in d:
        flags.append("⚠️ Не работает в других странах")
    if "регион" in d and "турция" in d:
        if "только" in d or "исключительно" in d:
            flags.append("⚠️ Явное региональное ограничение")
    return flags

def llm_check(description, buyer_platform, nominal):
    """ИИ-проверка: может ли эта карта быть продана на скупщике."""
    if not HAS_LLM:
        return {"verdict": "UNKNOWN", "reason": "LLM не подключен"}
    
    buyer_info = BUYER_REGIONS.get(buyer_platform, "Информации о региональных ограничениях нет")
    
    prompt = f"""Описание товара с Plati.Market:
{description}

Товар: Steam Turkey {nominal} USD gift card.
Покупатель (скупщик): {buyer_platform} — {buyer_info}

Вопрос: 
1. Для какого региона эта карта? (Steam Turkey, USD-аккаунт)
2. Можно ли её продать на {buyer_platform}? Есть ли риск, что {buyer_platform} откажется её принять из-за региональных ограничений?

Ответ строго:
ВЕРДИКТ: SAFE или RISKY или INCOMPATIBLE
РЕГИОН_КАРТЫ: <регион>
ПРИЧИНА: <1 предложение>"""
    
    try:
        ans = ask(prompt, system_prompt="Ты анализируешь совместимость подарочных карт. Отвечай кратко, по шаблону.")
    except Exception as e:
        return {"verdict": "UNKNOWN", "reason": f"LLM ошибка: {e}"}
    
    verdict = "UNKNOWN"
    region = "?"
    reason = ""
    for line in ans.split("\n"):
        if "ВЕРДИКТ:" in line:
            v = line.split(":", 1)[1].strip().upper()
            if "SAFE" in v: verdict = "SAFE"
            elif "RISKY" in v: verdict = "RISKY"
            elif "INCOMPATIBLE" in v: verdict = "INCOMPATIBLE"
        elif "РЕГИОН_КАРТЫ:" in line:
            region = line.split(":", 1)[1].strip()
        elif "ПРИЧИНА:" in line:
            reason = line.split(":", 1)[1].strip()
    
    return {"verdict": verdict, "region": region, "reason": reason, "raw": ans[:300]}

def check_bundle(product_url, buyer_platform, nominal):
    """Полная проверка: правила + ИИ."""
    desc = fetch_description(product_url)
    if not desc:
        return {"verdict": "UNKNOWN", "reason": "Не удалось загрузить описание"}
    
    flags = rule_based_check(desc)
    llm_result = llm_check(desc, buyer_platform, nominal)
    
    # Комбинируем
    final_verdict = llm_result["verdict"]
    if flags and final_verdict == "SAFE":
        final_verdict = "RISKY"
    
    return {
        "verdict": final_verdict,
        "region": llm_result.get("region", "?"),
        "reason": llm_result.get("reason", ""),
        "flags": flags,
        "buyer": buyer_platform,
    }

if __name__ == "__main__":
    # Тест: проверяем текущую связку

        "https://plati.market/itm/4377178",

        10
    )

    print(f"Регион:    {result['region']}")
    print(f"Причина:   {result['reason']}")

        "query": query,

def find_best_lot(items, target=10):
    """Фильтрует по минимальному номиналу = target."""
    candidates = []
    for item in items:
        name = (item.get("name_eng","") + " " + item.get("name","")).lower()

        url = item.get("url")
        if not price or not url: continue

            "seller": item.get("seller_name", ""),
            "url": url,
            "name": item.get("name", "")[:60],
        })

    return candidates

def scan():

def fetch_plati_lots(nominal):
    """Все доступные лоты на Plati для номинала (где nominal = минимальный)."""
    params = {
        "query": f"Steam Turkey {nominal} USD",
        "pagesize": 50, "pagenum": 1,
        "visibleOnly": "true", "response": "json"
    }
    try:
        r = requests.get(API, params=params, headers=HEADERS, timeout=20)
        items = r.json().get("items", [])
    except Exception as e:
        print(f"  ⚠️ API error {nominal}: {e}")
        return []
    
    lots = []
    for item in items:
        name = item.get("name_eng", "").lower() + " " + item.get("name", "").lower()
        price = item.get("price_usd")
        url = item.get("url")
        if not price or not url: continue
        m = re.search(r'(\d+)', name)
        if not m: continue
        if int(m.group(1)) != nominal: continue
        if price < nominal * 0.4: continue  # отсев фейков
        lots.append({
            "nominal": nominal,
            "price": float(price),
            "seller": item.get("seller_name", ""),
            "url": url,
            "name": item.get("name", "")[:50],
        })
    lots.sort(key=lambda x: x["price"])
    return lots

def fetch_fazercards_turkey():
    """Публичные цены FazerCards Turkey (HTML)."""
    try:
        r = requests.get("https://reseller.fazercards.com/en/catalog/steam/turkey", 
                         headers=HEADERS, timeout=20)
        if r.status_code != 200: return []
        soup = BeautifulSoup(r.text, "html.parser")
        text = soup.get_text(separator="\n", strip=True)
        # Ищем пары "X USD $Y"
        pairs = re.findall(r'(\d+)\s*USD\s*\$\s*([\d.]+)', text)

            nominal = int(nominal_str)

            key = (nominal, price)
            if key in seen: continue
            seen.add(key)
            # Применяем скидку Gold (3.55%) — предполагаем, что она есть
            price_gold = round(price * 0.9645, 2)

                "price": price_gold,
                "price_public": price,
                "url": "https://reseller.fazercards.com/en/catalog/steam/turkey",

def parse_fazercards_region(region_name, url):
    """Парсит пакеты: "X USD $Y" — номинал X USD, цена Y USD."""
    html = fetch(url)
    if not html:
        return []
